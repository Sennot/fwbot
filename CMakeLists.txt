cmake_minimum_required(VERSION 3.25)
project(FrameWindowLab VERSION 1.0.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(FWL_BUILD_MOD "Build the Windows Geode mod" ON)
option(FWL_BUILD_TESTS "Build portable parser and analysis tests" ON)

add_library(fwl_core STATIC
    src/core/Analysis.cpp
    src/core/ReplayIO.cpp
    src/core/Report.cpp
)
target_include_directories(fwl_core PUBLIC src/core PRIVATE vendor)
target_compile_definitions(fwl_core PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
    target_compile_options(fwl_core PRIVATE /EHsc /utf-8 /W4)
else()
    target_compile_options(fwl_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(FWL_BUILD_TESTS)
    enable_testing()
    add_executable(fwl_tests tests/core_tests.cpp)
    target_link_libraries(fwl_tests PRIVATE fwl_core)
    target_include_directories(fwl_tests PRIVATE vendor)
    target_compile_definitions(fwl_tests PRIVATE FWL_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
    if(MSVC)
        target_compile_options(fwl_tests PRIVATE /EHsc /utf-8)
    endif()
    add_test(NAME parser_and_analysis COMMAND fwl_tests)
endif()

if(FWL_BUILD_MOD)
    if(NOT DEFINED ENV{GEODE_SDK})
        message(FATAL_ERROR "Set GEODE_SDK to the Geode 5.8.2 SDK, or use -DFWL_BUILD_MOD=OFF for core tests.")
    endif()
    if(NOT WIN32 AND NOT GEODE_TARGET_PLATFORM STREQUAL "Win64")
        message(FATAL_ERROR "This mod targets Windows x64 only.")
    endif()
    add_library(${PROJECT_NAME} SHARED src/Engine.cpp src/Hooks.cpp src/LabPopup.cpp)
    target_compile_definitions(${PROJECT_NAME} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    add_subdirectory($ENV{GEODE_SDK} ${CMAKE_CURRENT_BINARY_DIR}/geode)
    target_link_libraries(${PROJECT_NAME} PRIVATE fwl_core)
    if(MSVC)
        target_compile_options(${PROJECT_NAME} PRIVATE /EHsc /utf-8)
    endif()
    # Match the keyword signature used above, including Geode's dependency links.
    setup_geode_mod(${PROJECT_NAME} LINK_TYPE PRIVATE)
endif()
